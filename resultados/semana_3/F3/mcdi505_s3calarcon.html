def calcular_exponente_hurst(serie, min_ventana=8, max_ventana=None, n_ventanas=10):
    """Exponente de Hurst por Rango Reescalado (R/S). Ver Sesión 3, §2.3."""
    serie = np.asarray(serie, dtype=float)
    n = len(serie)
    if max_ventana is None:
        max_ventana = n // 4
    ventanas = np.unique(np.geomspace(min_ventana, max_ventana, num=n_ventanas, dtype=int))
    rs_promedio, ventanas_validas = [], []
    for w in ventanas:
        m = n // w
        if m < 1:
            continue
        rs_bloques = []
        for i in range(m):
            bloque = serie[i * w:(i + 1) * w]
            recorrido = np.cumsum(bloque - np.mean(bloque))
            desviacion = np.std(bloque)
            if desviacion > 0:
                rs_bloques.append((np.max(recorrido) - np.min(recorrido)) / desviacion)
        if rs_bloques:
            rs_promedio.append(np.mean(rs_bloques))
            ventanas_validas.append(w)
    pendiente, _ = np.polyfit(np.log(ventanas_validas), np.log(rs_promedio), 1)
    return pendiente


def aplicar_filtro_fraccional(serie, d):
    """Operador (1-B)^d. d > 0 diferencia fraccionalmente; d < 0 integra. Ver Sesión 3, §2.3."""
    serie = np.asarray(serie, dtype=float)
    n = len(serie)
    pesos = np.zeros(n)
    pesos[0] = 1.0
    for k in range(1, n):
        pesos[k] = pesos[k - 1] * (k - 1 - d) / k
    return np.array([np.sum(pesos[:t + 1] * serie[t::-1]) for t in range(n)])


def garch_log_verosimilitud_negativa(parametros, serie):
    """Log-verosimilitud negativa de un GARCH(1,1) gaussiano. Ver Sesión 3, §3.4."""
    omega, alpha, beta = parametros
    sigma2 = np.zeros(len(serie))
    sigma2[0] = np.var(serie)
    for t in range(1, len(serie)):
        sigma2[t] = omega + alpha * serie[t - 1] ** 2 + beta * sigma2[t - 1]
    if np.any(sigma2 <= 1e-8):
        return 1e10
    return 0.5 * np.sum(np.log(2 * np.pi) + np.log(sigma2) + serie ** 2 / sigma2)


def correccion_arfima(residuos, d, h):
    """Pronóstico h pasos del componente ARFIMA(0,d,0) de los residuos.

    Diferencia fraccionalmente los residuos con d, pronostica cero (ruido blanco)
    y vuelve a integrar con -d. Lo que queda es la persistencia que la memoria
    larga añadiría al pronóstico del SARIMA (en escala logarítmica).
    """
    r = np.asarray(residuos, dtype=float)
    extendida = np.concatenate([aplicar_filtro_fraccional(r, d), np.zeros(h)])
    return aplicar_filtro_fraccional(extendida, -d)[len(r):]


def estimar_garch11(residuos, h):
    """Estima un GARCH(1,1) por MLE sobre los residuos estandarizados.

    Devuelve un diccionario con omega, alpha, beta, persistencia, convergencia,
    residuos estandarizados por sigma_t y el factor de escala de la desviación
    estándar condicional para los h meses siguientes (1 = igual a la varianza
    incondicional).
    """
    r = np.asarray(residuos, dtype=float)
    s = r.std()
    u = r / s
    res = minimize(garch_log_verosimilitud_negativa, x0=[0.05, 0.10, 0.80], args=(u,),
                   bounds=[(1e-6, 5.0), (1e-6, 0.999), (1e-6, 0.999)],
                   constraints=({"type": "ineq", "fun": lambda p: 0.999 - p[1] - p[2]},),
                   method="SLSQP")
    om, al, be = res.x
    sigma2 = np.zeros(len(u)); sigma2[0] = u.var()
    for t in range(1, len(u)):
        sigma2[t] = om + al * u[t - 1] ** 2 + be * sigma2[t - 1]
    futura = np.zeros(h)
    futura[0] = om + al * u[-1] ** 2 + be * sigma2[-1]
    for i in range(1, h):
        futura[i] = om + (al + be) * futura[i - 1]
    incondicional = om / (1 - al - be)
    return {"omega": om, "alpha": al, "beta": be, "persistencia": al + be,
            "convergencia": bool(res.success),
            "residuos_estandarizados": u / np.sqrt(sigma2),
            "escala_futura": np.sqrt(futura / incondicional)}


def metricas(real, pred):
    """RMSE y MAE en la escala de los datos (GWh)."""
    e = np.asarray(real, float) - np.asarray(pred, float)
    return {"RMSE": float(np.sqrt(np.mean(e ** 2))), "MAE": float(np.mean(np.abs(e)))}


def cobertura_y_ancho(real, inferior, superior):
    """Proporción de meses dentro del intervalo y ancho promedio del intervalo (GWh)."""
    real, inferior, superior = (np.asarray(x, float) for x in (real, inferior, superior))
    return {"Cobertura": float(np.mean((real >= inferior) & (real <= superior))),
            "Ancho medio": float(np.mean(superior - inferior))}

Punto de partida: SARIMA de la Sumativa 2¶

El modelo seleccionado en la Sumativa 2 es M4 = SARIMA(0,1,0)(1,1,0)₁₂ sobre log(consumo), sin variables exógenas (M6, con covid y agudo, fue solo un análisis complementario y no participa aquí). Se reestima con la misma especificación y sobre el mismo entrenamiento de la Sumativa 2 (2015-01 a 2022-12), con la convención de signos de statsmodels:

$$(1-\Phi_1 B^{12})(1-B)(1-B^{12})\,\log y_t = \varepsilon_t .$$

El coeficiente reestimado (Φ1 = −0,311; σ² = 0,00193) coincide con el de la Sumativa 2 (−0,311), y la diferencia de log-verosimilitud es nula a tres decimales. El AIC es −277,8. En la validación 2023, RMSE = 4,12 y MAE = 3,71 GWh, las mismas cifras de la Sumativa 2.

Se descartan los primeros d + D·s = 13 residuos. Los residuos de arranque del filtro de Kalman que superan 3,5 s están en ese tramo (t = 0: 105 s; t = 12: −52 s), y ninguno posterior lo hace. Quedan 83 residuos (2016-02 a 2022-12).

1. Modelamiento de memoria larga (ARFIMA / diferenciación fraccional)¶

Precaución metodológica. Con solo 83 residuos del SARIMA M4, el estimador R/S del exponente de Hurst (Hurst, 1951) está sesgado hacia arriba, por lo que H > 0,5 no basta para concluir memoria larga. Por eso H se contrasta con ruido blanco simulado del mismo tamaño, con la ACF residual hasta el rezago 36 y con el efecto de ARFIMA(0,d,0) sobre el pronóstico de 2023.

Conclusión. No hay evidencia suficiente de memoria larga relevante. Por esta razón, no se justifica mantener ARFIMA en el modelo final.

Conclusión. GARCH queda como modelo de contraste, no como modelo necesario.

  • Memoria larga. La ACF residual no decae lentamente (1 de 36 rezagos fuera de banda). H = 0,584 y d = 0,084 son bajos, y el 60,5 % de los ruidos blancos con n = 83 supera ese H. Con ARFIMA, Ljung-Box a 24 rezagos empeora (p de 0,059 a 0,023).
  • Volatilidad. Ljung-Box de residuos² (p = 0,829) y ARCH-LM (p = 0,786) no rechazan. El GARCH(1,1) tiene α + β = 0,8403, pero α = 0,0687 es pequeño y los residuos estandarizados no muestran estructura (p = 0,766 y 0,854).
  • Pronóstico 2023 (RMSE / MAE, GWh). SARIMA M4: 4,12 / 3,71. Con ARFIMA: 4,19 / 3,75. Naive estacional: 7,26 / 6,53.
  • Intervalos 2023. Cobertura 12/12 en ambos casos; ancho medio de 46,26 GWh (SARIMA) y 45,64 GWh (GARCH).
  • Parsimonia. M4 estima un solo coeficiente (Φ1 = −0,311). ARFIMA agrega d y GARCH agrega ω, α y β.

Decisión. Se mantiene M4 = SARIMA(0,1,0)(1,1,0)₁₂ (en statsmodels: order=(0,1,0), seasonal_order=(1,1,0,12)) como modelo para la Semana 4. ARFIMA no mejora la predicción, no existe evidencia suficiente de efectos ARCH y GARCH no produce una mejora sustantiva de los intervalos. M4 presenta buen desempeño predictivo y menor complejidad. Como limitación, proyecta 1325,4 GWh para 2023 frente a 1363,9 GWh observados (−2,8 %) y subestima 9 de 12 meses (Fig. 7).

Conclusiones¶

  1. Qué se evaluó. Se evaluó si la memoria larga (ARFIMA) o la volatilidad condicional (GARCH) aportan al modelo seleccionado en la Sumativa 2, M4 = SARIMA(0,1,0)(1,1,0)₁₂ sobre log(consumo). Se usaron los 83 residuos del entrenamiento 2015-01 a 2022-12 y la validación de 2023.
  2. Memoria larga. No hay evidencia suficiente de memoria larga relevante. H = 0,584 (d = 0,084) es compatible con lo que produce el ruido blanco con 83 observaciones, la ACF no decae lentamente y la corrección ARFIMA empeora levemente el pronóstico (RMSE de 4,12 a 4,19 GWh).
  3. Volatilidad. No hay evidencia estadística suficiente de heterocedasticidad condicional. Ljung-Box sobre los cuadrados (p = 0,829) y ARCH-LM (p = 0,786) no rechazan, y los residuos estandarizados no muestran problemas. La persistencia α + β = 0,8403 no basta, por sí sola, para afirmar efectos ARCH.
  4. Comparación y selección. El SARIMA M4 obtiene el menor error en 2023 (RMSE 4,12 y MAE 3,71 GWh) y supera ampliamente al naive estacional (7,26 y 6,53). Las bandas GARCH mantienen la cobertura de 12/12 y reducen poco el ancho medio (de 46,26 a 45,64 GWh). Por esta razón, se mantiene M4 para la Semana 4.
  5. Limitaciones. La muestra residual es corta (83 valores) y la validación cubre solo 12 meses, por lo que las diferencias pequeñas deben tomarse con precaución. Los resultados son propios de esta serie y 2024 no se utilizó. M4 subestima 9 de 12 meses de 2023 y la ACF residual conserva un rezago fuera de banda (24). Además, la ausencia de evidencia no demuestra que no exista memoria larga o heterocedasticidad; solo indica que estos datos no permiten detectarlas.
